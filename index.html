Rich flower drawing functions def draw_luxury_flower(ax, x, y, size, petal_color, center_color=(0.55, 0.27, 0.07), petals=8, zorder=20): for i in range(petals): angle = 2 * np.pi * i / petals px = x + size * 0.5 * np.cos(angle) py = y + size * 0.5 * np.sin(angle) ellipse = Ellipse((px, py), size*0.7, size*0.35, angle=np.degrees(angle), facecolor=petal_color, edgecolor=(0.6, 0.2, 0.2), linewidth=0.5, alpha=0.85, zorder=zorder) ax.add_patch(ellipse) center = Circle((x, y), size*0.2, facecolor=center_color, edgecolor=(0.3, 0.15, 0.05), linewidth=0.5, zorder=zorder+1) ax.add_patch(center) highlight = Circle((x-size*0.05, y+size*0.05), size*0.08, facecolor=(1, 0.9, 0.5, 0.6), edgecolor='none', zorder=zorder+2) ax.add_patch(highlight) def draw_gold_leaf(ax, x, y, size, angle, zorder=19): rad = np.radians(angle) dx = size * np.cos(rad) dy = size * np.sin(rad) perp_x = -dy * 0.4 perp_y = dx * 0.4 verts = [(x, y), (x + dx*0.5 + perp_x, y + dy*0.5 + perp_y), (x + dx, y + dy), (x + dx*0.5 - perp_x, y + dy*0.5 - perp_y), (x, y)] codes = [Path.MOVETO, Path.CURVE4, Path.CURVE4, Path.CURVE4, Path.CLOSEPOLY] leaf = PathPatch(Path(verts, codes), facecolor=(0.85, 0.65, 0.13, 0.7), edgecolor=(0.7, 0.5, 0.1), linewidth=1, zorder=zorder) ax.add_patch(leaf) ax.plot([x, x+dx], [y, y+dy], color=(0.9, 0.75, 0.2), linewidth=0.8, alpha=0.8, zorder=zorder+1) #
